# FILE: _includes/ordinal-date.html # PURPOSE: Outputs a date as "DDth Month YYYY" (e.g. "22nd September 2026"), # matching the date format used in figure captions across the site # (Videos, Photos, KFI). Jekyll/Liquid has no built-in ordinal-suffix # filter, so this computes it by hand from the day-of-month. # USAGE: {% include ordinal-date.html date=some_date_value %} # WARNING: do NOT call this from inside a for loop -- on this # site's Liquid version, repeatedly including the same file inside a # loop trips a Liquid::StackLevelError ("Nesting too deep"), even # though nothing is actually recursive. videos.html hit exactly this # and now inlines the same case/when logic directly in its loop # instead of using this include. Safe to use here for a one-off date # outside any loop. {%- assign day = include.date | date: "%-d" -%} {%- case day -%} {%- when '1' or '21' or '31' -%} {%- assign suffix = 'st' -%} {%- when '2' or '22' -%} {%- assign suffix = 'nd' -%} {%- when '3' or '23' -%} {%- assign suffix = 'rd' -%} {%- else -%} {%- assign suffix = 'th' -%} {%- endcase -%} {{- day -}}{{- suffix -}} {{- include.date | date: "%B %Y" -}}
